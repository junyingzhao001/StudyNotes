cmake_minimum_required(VERSION 3.16)
project(CppQuickStart LANGUAGES CXX)
find_package(Threads REQUIRED)

function(configure_course_target target)
    target_compile_features(${target} PRIVATE cxx_std_17)
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endfunction()

foreach(number 00 01 02 03 04 05 06 07 08 09 10 11 14 17 18 19 20 21)
    add_executable(demo_${number} demos/${number}.cpp)
    configure_course_target(demo_${number})
endforeach()
foreach(number 17 18 19 20 21)
    target_link_libraries(demo_${number} PRIVATE Threads::Threads)
endforeach()
